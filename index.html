document.addEventListener("DOMContentLoaded", function () { fetch("./Resources/r_index.json") .then((response) => { if (!response.ok) { throw new Error("Network response was not ok"); } return response.json(); }) .then((data) => { populateContent(data); }) .catch((error) => { console.error("Error loading r_index.json:", error); }); }); function populateContent(data) { const container = document.getElementById("container"); if (!data || data.length === 0) { document.getElementById("no_progs_div").classList.remove("no_progs_hidden"); return; } data.forEach((item) => { const pane = document.createElement("div"); pane.className = "pane"; const questionDiv = document.createElement("div"); questionDiv.className = "question"; const questionHeading = document.createElement("h1"); questionHeading.className = "qnH"; questionHeading.textContent = item.question || "No question provided"; questionDiv.appendChild(questionHeading); const btnDiv = document.createElement("div"); btnDiv.className = "dwnld_btn"; if (item.link) { const openLink = document.createElement("a"); openLink.href = item.link; openLink.target = "_blank"; openLink.className = "open_a"; const openButton = document.createElement("button"); openButton.className = "open_btn"; openButton.textContent = "Open"; openLink.appendChild(openButton); btnDiv.appendChild(openLink); } if (item.download) { const downloadLink = document.createElement("a"); downloadLink.href = item.download; downloadLink.className = "download_a"; downloadLink.download = ""; const downloadButton = document.createElement("button"); downloadButton.className = "download_btn"; const downloadImg = document.createElement("img"); downloadImg.src = "./Resources/download_icon.png"; downloadImg.alt = "Download"; downloadButton.appendChild(downloadImg); downloadLink.appendChild(downloadButton); btnDiv.appendChild(downloadLink); } pane.appendChild(questionDiv); pane.appendChild(btnDiv); container.appendChild(pane); }); }
