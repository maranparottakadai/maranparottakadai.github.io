# List of fruits
fruits = ["Apple", "Banana", "Orange", "Grapes"]
print("Fruits List:", fruits)

# Get the length of the list
l = len(fruits)
print("Length of the List:", l)

# Add a new fruit to the list
fruits.append("Cherry")
print("Updated Fruits List:", fruits)

# Get the length of the updated list
l = len(fruits)
print("Length of the List:", l)

# Sort the fruits list alphabetically
fruits.sort()
print("Sorted Fruits List:", fruits)

# Search for 'Pineapple' in the fruits list
print("Search 'Pineapple' in fruits list:")
s = "Pineapple" in fruits
print(s)

# Search for 'Apple' in the fruits list
print("Search 'Apple' in fruits list:")
s = "Apple" in fruits
print(s)

# Delete the fruit at index 3 (originally 'Grapes')
print("Fruits List after deleting", fruits[3], ":")
del fruits[3]
print(fruits)
